/* programa para inverter um vetor com dez elementos. */

int vet[10];

int invert(int v[], int inicio, int fim)
{
    int temp;

    if(inicio >= fim)
        return 1;
    else
    {
        temp = v[inicio];
        v[inicio] = v[fim];
        v[fim] = temp;

        invert(v, inicio + 1, fim - 1);
    }
}

void main(void)
{
    int i;

    i = 0;
    while(i < 10)
    {
        vet[i] = input();
        i = i + 1;
    }

    invert(vet, 0, 9);

    i = 0;
    while(i < 10)
    {
        output(vet[i]);
        i = i + 1;
    }
}
