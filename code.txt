/* programa para busca em um vetor */

int vet[10];

int busca(int v[], int inicio, int fim, int x)
{
    int meio;

    if(inicio > fim)
    {
        return 11;
    }
    else
    {
        meio = (inicio + fim) / 2;

        if(v[meio] == x)
        {
            return meio;
        }
        else
        {
            if(x < v[meio])
            {
                return busca(v, inicio, meio - 1, x);
            }
            else
            {
                return busca(v, meio + 1, fim, x);
            }
        }
    }
}

void main(void)
{
    int i;
    int x;
    int pos;

    i = 0;

    while(i < 10)
    {
        vet[i] = input();
        i = i + 1;
    }

    x = input();

    pos = busca(vet, 0, 9, x);

    output(pos);
}
