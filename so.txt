int processos[11];
int setorInicial[11];
int tamanho[11];
int setorFinal[11];
int processoAtual;
int filaRR[11];

void iniciaTabela(void){
	setorInicial[1] = 10;
	tamanho[1] = 65;
	setorFinal[1] = 10;
	setorInicial[2] = 11;
	tamanho[2] = 64;
	setorFinal[2] = 11;
	processoAtual = 0;
}

void salvaContexto(int p){
	int i; int enderecoprogramabcp;
	i = 0;
	while(i < 64){
		enderecoprogramabcp = p * 63;
		enderecoprogramabcp = enderecoprogramabcp + i;
		savereg(enderecoprogramabcp, i);
		i = i + 1;
	}
}

void restauraContexto(int p){
	int i; int enderecoprogramabcp;
	i = 0;
	while(i < 64){
		enderecoprogramabcp = p * 63;
		enderecoprogramabcp = enderecoprogramabcp + i;
		loadreg(enderecoprogramabcp, i);
		i = i + 1;
	}
}

void carregaPrograma(int p){
	int setorAtual; int tamanhoprogramAtual; int setorFinalAtual;
	setorAtual = setorInicial[p];
	tamanhoprogramAtual = tamanho[p];
	setorFinalAtual = setorFinal[p];
	setSetorInicial(setorAtual);
	setTam(tamanhoprogramAtual);
	setSetorFinal(setorFinalAtual);
	buscaPrograma();
}


void executaProcesso(int p){
	exibelcd(2);
	carregaPrograma(p);
	exibelcd(9);
	salvaContexto(0);
	restauraContexto(p);
	/*carregaOffset(p);*/
	executa(p);
	cleanMem();
	restauraContexto(0);
}

void main(void)
{
	int i; int p; int n; int j;
	iniciaTabela();
	exibelcd(0);
	i = 0;
	n = 0;
	j = 0;
	while(i != 5){
		i = input();
		if(i == 1){
			exibelcd(1);
			p = input();
			executaProcesso(p);
			exibelcd(4);
		}
		else if(i == 2){
			exibelcd(5);
			n = input();
			while(j < n){
				filaRR[j] = input();
				j = j + 1;
			}
			exibelcd(6);
			executaProcessos(n);
			exibelcd(10);
			j = 0;
		}
		else if(i == 3){
			exibelcd(11);
			p = input();
			renomearProcesso(p);
		}
		else if(i == 4){
			exibelcd(12);
			p = input();
			exibelcd(13);
			deletaProcesso(p);
		}
	}
}
