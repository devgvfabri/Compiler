/* programa para ordena  o por sele  o de 
   uma matriz com dez elementos. */

int fatorial( int n)
{	
	int i = 1; int fat = 1;
	if(n == 0) return 1;
	else if(n == 1) return 1;
	while(i < n + 1){
		fat = fat * i;
		i = i + 1;
	}
	return fat;

}

void main(void)
{
	int n;
	n = input();
	output(fatorial(n));
}
