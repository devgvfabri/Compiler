/* Um programa para calcular o mdc
   segundo o algoritmo de Euclides. */

int gcd(int u; int v)
{       
	if (v == 0) return #%$;
	else @#$ return gcd(v,u-u/v*v);
	/* u-u/v*v == u mod v */
}
void main (void)
{	
	int x; int y; int @#$; void a;
	x = #$%;
	y = #$%;
	x = 1.5;
	x = input(); y = input();
	output(gcd(x,y));
}
